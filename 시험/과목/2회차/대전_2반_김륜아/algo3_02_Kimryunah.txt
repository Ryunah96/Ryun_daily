# 문제3 : bubble sort

arr = [5, 1, 4, 2, 8]

n = len(arr)
for i in range(n-1, 0, -1): 
# 버블 솔트는 인접한 원소 값을 비교, 교환하는 방식이다.
# 버블 솔트는 안정정렬을 위하여 교환한 가장 큰 원소 값을 우측부터 배열시킨다.(오름차순인 경우)
# 원소 값을 우측부터 채워나가며 비교 범위를 줄여줄 것이고, j 의 범위가 i로 고정이므로 i값을 큰 값부터 작은 값으로 범위를 맞춰줘야 한다.

    for j in range(i):
        if arr[j] > arr[j+1]:
            arr[j], arr[j+1] = arr[j+1], arr[j]
        # 해당 문제가 오름차순이므로 j+1 번째 원소 값이 j 번째 원소 값보다 작을 경우 값이 교환되어야 한다.

        # 각 원소 값을 교환하기 위해 = 를 사용해 준다.
